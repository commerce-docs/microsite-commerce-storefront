### Header example

This example shows how to customize the `Header` slot:

```js
import { render as provider } from '@dropins/storefront-requisition-list/render.js';
import RequisitionListGrid from '@dropins/storefront-requisition-list/containers/RequisitionListGrid.js';

provider.render(RequisitionListGrid, {
  slots: {
    Header: (ctx) => {
      const customHeader = document.createElement('div');
      customHeader.className = 'custom-requisition-list-header';
      customHeader.textContent = 'My Custom Requisition Lists Header';
      ctx.replaceWith(customHeader);
    }
  }
})(document.querySelector('.requisition-list-grid'));
```
