## How to customize

Override any keys above by creating a custom dictionary file:

```javascript title="scripts/config/dictionaries/recommendations-dictionary.js"
export const recommendationsDictionary = {
  "Recommendations": {
    "ProductList": {
      "addToCart": "Custom string",
      "selectOptions": "Custom string"
    }
  }
};
```

Then import it in your initializer:

```javascript title="scripts/initializers/recommendations.js"
import { initializers } from '@dropins/tools/initializer.js';
import { initialize } from '@dropins/storefront-recommendations';
import { recommendationsDictionary } from '../config/dictionaries/recommendations-dictionary.js';

const langDefinitions = {
  default: {
    ...recommendationsDictionary,
  },
};

await initializers.mountImmediately(initialize, { langDefinitions });
```
