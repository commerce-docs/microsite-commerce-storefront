## How to customize

Override any keys above by creating a custom dictionary file:

```javascript title="scripts/config/dictionaries/wishlist-dictionary.js"
export const wishlistDictionary = {
  "Wishlist": {
    "EmptyWishlist": {
      "heading": "Custom string",
      "message": "Custom string"
    }
  }
};
```

Then import it in your initializer:

```javascript title="scripts/initializers/wishlist.js"
import { initializers } from '@dropins/tools/initializer.js';
import { initialize } from '@dropins/storefront-wishlist';
import { wishlistDictionary } from '../config/dictionaries/wishlist-dictionary.js';

const langDefinitions = {
  default: {
    ...wishlistDictionary,
  },
};

await initializers.mountImmediately(initialize, { langDefinitions });
```
