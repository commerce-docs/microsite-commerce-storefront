```js
// 1. Import initializer (handles all setup)
import '../../scripts/initializers/company-switcher.js';

// 2. Import the API functions you need

  getCustomerCompanyInfo,
  setCompanyHeaders 
} from '@dropins/storefront-company-switcher/api.js';

// 3. Get current company context
```
