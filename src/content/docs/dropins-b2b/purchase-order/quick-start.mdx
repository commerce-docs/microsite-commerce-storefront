### Quick example

This example shows the basic pattern for using the Purchase Order drop-in. For complete details and advanced patterns, see the [Using drop-ins](/dropins/all/quick-start/) guide.

```js
// 1. Import initializer (handles all setup)
import '../../scripts/initializers/purchase-order.js';

// 2. Import the container you need
import ApprovalRuleDetails from '@dropins/storefront-purchase-order/containers/ApprovalRuleDetails.js';

// 3. Import the provider
import { render as provider } from '@dropins/storefront-purchase-order/render.js';

// 4. Render in your block
export default async function decorate(block) {
  await provider.render(ApprovalRuleDetails, {
    // Configuration options - see Containers page
  })(block);
}
```
